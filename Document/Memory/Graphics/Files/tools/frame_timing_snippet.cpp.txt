// (A) DirectX12/DirectX12API.cpp — include 목록 끝
#include <chrono>   // 임시 계측 (5-5 전후 비교)

// (B) DirectX12/DirectX12API.cpp — PDirectX12API::BeginFrame() 전체를 아래로
void PDirectX12API::BeginFrame()
{
	// ---- 임시 계측 (5-5 전후 비교): 프레임 시간과 Begin()의 펜스 대기 시간을 600프레임마다 찍는다. 검증 뒤 제거 ----
	using HTimingClock = std::chrono::steady_clock;
	static HTimingClock::time_point s_lastBeginStart;
	static float64 s_frameMsSum = 0.0;
	static float64 s_frameMsMax = 0.0;
	static float64 s_waitMsSum  = 0.0;
	static float64 s_waitMsMax  = 0.0;
	static uint64  s_windowFrames = 0;
	static uint64  s_totalFrames  = 0;
	const HTimingClock::time_point beginStart = HTimingClock::now();
	// ----

	_commandQueue->Begin();

	// ---- 임시 계측 ----
	const float64 waitMs = std::chrono::duration<float64, std::milli>(HTimingClock::now() - beginStart).count();
	if (s_totalFrames > 0)
	{
		const float64 frameMs = std::chrono::duration<float64, std::milli>(beginStart - s_lastBeginStart).count();
		s_frameMsSum += frameMs;
		s_frameMsMax  = std::max(s_frameMsMax, frameMs);
		s_waitMsSum  += waitMs;
		s_waitMsMax   = std::max(s_waitMsMax, waitMs);
		++s_windowFrames;
		if (s_windowFrames == 600)
		{
			JG_LOG(Graphics, ELogLevel::Info, "FrameTiming : frames %d-%d, frame avg %.3f ms (max %.3f, %.0f fps), fence wait avg %.3f ms (max %.3f)",
				(int32)(s_totalFrames - 599), (int32)s_totalFrames, s_frameMsSum / 600.0, s_frameMsMax, 600000.0 / s_frameMsSum, s_waitMsSum / 600.0, s_waitMsMax);
			s_frameMsSum = 0.0;
			s_frameMsMax = 0.0;
			s_waitMsSum  = 0.0;
			s_waitMsMax  = 0.0;
			s_windowFrames = 0;
		}
	}
	s_lastBeginStart = beginStart;
	++s_totalFrames;
	// ----

	_resourceStagingManager->BeginFrame();   // 지난 제출의 리드백 완료 처리(콜백은 여기서, 메인 스레드)
}
