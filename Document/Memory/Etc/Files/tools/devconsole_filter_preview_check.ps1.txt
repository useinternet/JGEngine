param(
    [Parameter(Mandatory = $true)][string]$Exe,
    [Parameter(Mandatory = $true)][string]$WorkDir,
    [Parameter(Mandatory = $true)][string]$OutPrefix,
    [int]$WaitSeconds = 20,
    # -Probe: 대기 뒤 캡처 한 장만 남기고 닫는다(좌표 확인용)
    [switch]$Probe,
    # 좌표는 창 캡처(PrintWindow) 기준 픽셀이다. 먼저 -Probe 캡처에서 읽는다.
    [int]$InputX = 0,
    [int]$InputY = 0,
    [int]$FilterX = 0,
    [int]$FilterY = 0,
    [int]$InfoX = 0,
    [int]$InfoY = 0
)

# DevConsole 로그 필터 · 명령 미리보기 창 입력 확인.
# 키·마우스 입력은 런처 창에만 PostMessage 로 보낸다(전경 창을 바꾸지 않으므로 다른 창에 입력이 들어가지 않는다).
# 엔진 창 프로시저는 모든 메시지를 ImGui_ImplWin32_WndProcHandler 로 넘긴다(DX12GUIBackend.cpp OnWndProc).
# 전제(검증 빌드 전용 임시 코드, devconsole_filter_preview_temp_snippets.cpp.txt (A)~(C)): DevConsole 이 시작 시 열리고,
#   Devkit 에 test.alpha / test.beta / test.gamma / demo.test 가 선언돼 있고, ImGui Win32 백엔드가 JG_UITEST=1 이면 보낸 마우스만 쓴다.
# 쓰는 법(2026-09-30): .ps1 로 복사할 때 UTF-8 BOM 을 유지한다(BOM 이 없으면 PowerShell 5.1 이 CP949 로 읽어 한글 주석 뒤 줄을 삼킨다).
#   1) -Probe 로 캡처 한 장 → 좌표 읽기  2) 좌표를 넘겨 전체 실행 → <OutPrefix>_1_list.png ... _10_level.png
# 결과 기록: ../2026-09-30_devconsole_filter_preview_verify.txt
$ErrorActionPreference = "Stop"
Add-Type -AssemblyName System.Drawing

Add-Type @"
using System;
using System.Runtime.InteropServices;
public class Win32PreviewCheck {
    [StructLayout(LayoutKind.Sequential)]
    public struct RECT { public int Left; public int Top; public int Right; public int Bottom; }
    [StructLayout(LayoutKind.Sequential)]
    public struct POINT { public int X; public int Y; }
    [DllImport("user32.dll")] public static extern bool GetWindowRect(IntPtr hWnd, out RECT lpRect);
    [DllImport("user32.dll")] public static extern bool PrintWindow(IntPtr hwnd, IntPtr hdcBlt, uint nFlags);
    [DllImport("user32.dll")] public static extern bool PostMessage(IntPtr hWnd, uint Msg, IntPtr wParam, IntPtr lParam);
    [DllImport("user32.dll")] public static extern bool ClientToScreen(IntPtr hWnd, ref POINT lpPoint);
}
"@

$WM_KEYDOWN = 0x0100
$WM_KEYUP   = 0x0101
$WM_CHAR    = 0x0102
$VK_BACK    = 0x08
$VK_TAB     = 0x09
$VK_RETURN  = 0x0D
$VK_UP      = 0x26
$VK_DOWN    = 0x28

function Save-Capture([IntPtr]$hwnd, [string]$path) {
    $rect = New-Object Win32PreviewCheck+RECT
    [Win32PreviewCheck]::GetWindowRect($hwnd, [ref]$rect) | Out-Null
    $bmp = New-Object System.Drawing.Bitmap(($rect.Right - $rect.Left), ($rect.Bottom - $rect.Top))
    $gfx = [System.Drawing.Graphics]::FromImage($bmp)
    $hdc = $gfx.GetHdc()
    $ok = [Win32PreviewCheck]::PrintWindow($hwnd, $hdc, 2)
    $gfx.ReleaseHdc($hdc)
    $gfx.Dispose()
    if (-not $ok) { Write-Output "PrintWindow failed" }
    $bmp.Save($path, [System.Drawing.Imaging.ImageFormat]::Png)
    $bmp.Dispose()
    Write-Output "Saved $path"
}

function Post-Click([IntPtr]$hwnd, [int]$windowX, [int]$windowY) {
    # 창 좌표(캡처 기준) → 클라이언트 좌표
    $rect = New-Object Win32PreviewCheck+RECT
    [Win32PreviewCheck]::GetWindowRect($hwnd, [ref]$rect) | Out-Null
    $origin = New-Object Win32PreviewCheck+POINT
    [Win32PreviewCheck]::ClientToScreen($hwnd, [ref]$origin) | Out-Null
    $x = $windowX - ($origin.X - $rect.Left)
    $y = $windowY - ($origin.Y - $rect.Top)
    $lParam = [IntPtr](($y -shl 16) -bor ($x -band 0xFFFF))
    # 실제 커서가 창 밖이면 백엔드의 TrackMouseEvent 가 곧 WM_MOUSELEAVE 를 보내므로 세 메시지를 잇달아 넣는다.
    [Win32PreviewCheck]::PostMessage($hwnd, 0x0200, [IntPtr]::Zero, $lParam) | Out-Null   # WM_MOUSEMOVE
    [Win32PreviewCheck]::PostMessage($hwnd, 0x0201, [IntPtr]1, $lParam) | Out-Null         # WM_LBUTTONDOWN
    [Win32PreviewCheck]::PostMessage($hwnd, 0x0202, [IntPtr]::Zero, $lParam) | Out-Null    # WM_LBUTTONUP
    Start-Sleep -Milliseconds 500
    Write-Output "Clicked client ($x, $y)"
}

function Post-Key([IntPtr]$hwnd, [int]$vk, [int]$count = 1) {
    for ($i = 0; $i -lt $count; $i++) {
        [Win32PreviewCheck]::PostMessage($hwnd, $WM_KEYDOWN, [IntPtr]$vk, [IntPtr]1) | Out-Null
        Start-Sleep -Milliseconds 80
        [Win32PreviewCheck]::PostMessage($hwnd, $WM_KEYUP, [IntPtr]$vk, [IntPtr]0xC0000001L) | Out-Null
        Start-Sleep -Milliseconds 150
    }
}

function Post-Text([IntPtr]$hwnd, [string]$text) {
    foreach ($c in $text.ToCharArray()) {
        [Win32PreviewCheck]::PostMessage($hwnd, $WM_CHAR, [IntPtr][int]$c, [IntPtr]1) | Out-Null
        Start-Sleep -Milliseconds 40
    }
    Start-Sleep -Milliseconds 300
}

function Shot([IntPtr]$hwnd, [string]$name) {
    Start-Sleep -Milliseconds 700
    Save-Capture $hwnd "$($OutPrefix)_$name.png"
}

# ImGui Win32 백엔드는 매 프레임 실제 커서 위치(WindowFromPoint(GetCursorPos()))로 마우스가 올라간 뷰포트를 정한다.
# 실제 커서가 런처 밖에 있으면 PostMessage 클릭은 허공 클릭이 되어 입력줄 포커스까지 풀린다.
# 검증 빌드의 백엔드 임시 코드가 JG_UITEST=1 일 때 보낸 마우스 메시지만 쓰게 한다(사용자 커서를 움직이지 않는다).
$env:JG_UITEST = "1"
$proc = Start-Process -FilePath $Exe -WorkingDirectory $WorkDir -PassThru
Write-Output "Started PID $($proc.Id)"

$hwnd = [IntPtr]::Zero
for ($i = 0; $i -lt 60; $i++) {
    Start-Sleep -Milliseconds 500
    $proc.Refresh()
    if ($proc.HasExited) { Write-Output "Process exited early with code $($proc.ExitCode)"; exit 2 }
    if ($proc.MainWindowHandle -ne [IntPtr]::Zero) { $hwnd = $proc.MainWindowHandle; break }
}
if ($hwnd -eq [IntPtr]::Zero) { Write-Output "No main window"; $proc.Kill(); exit 3 }
Write-Output "Main window: $hwnd"

Start-Sleep -Seconds $WaitSeconds
$proc.Refresh()
if ($proc.HasExited) { Write-Output "Process exited during wait with code $($proc.ExitCode)"; exit 4 }

if ($Probe) {
    Shot $hwnd "0_probe"
}
else {
    # 1) 이름을 치는 중: 후보 목록(가장 잘 맞는 test.alpha 가 입력줄 바로 위, 이름만 'test' 를 품은 demo.test 가 맨 위)
    Post-Click $hwnd $InputX $InputY
    Post-Text $hwnd "test"
    Shot $hwnd "1_list"

    # 2) ↑ 두 번: test.beta 선택, 입력줄 = test.beta
    Post-Key $hwnd $VK_UP 2
    Shot $hwnd "2_select"

    # 3) Tab: "test.beta " 로 완성 → 사용법 모드
    Post-Key $hwnd $VK_TAB
    Shot $hwnd "3_usage"

    # 4) 인자를 붙여 실행
    Post-Text $hwnd "hello"
    Post-Key $hwnd $VK_RETURN
    Start-Sleep -Milliseconds 700

    # 5) 없는 이름 + 공백: 빨간 Unknown 줄. 실행해서 오류 줄도 남긴다(필터 확인용)
    Post-Text $hwnd "nosuch x"
    Shot $hwnd "4_unknown"
    Post-Key $hwnd $VK_RETURN
    Start-Sleep -Milliseconds 700

    # 6) 후보 없음: 'zzz'. 이때 ↑ 는 히스토리 → 가장 최근 줄 "nosuch x"(사용법 모드: Unknown)
    Post-Text $hwnd "zzz"
    Shot $hwnd "5_nomatch"
    Post-Key $hwnd $VK_UP
    Shot $hwnd "6_history_unknown"
    Post-Key $hwnd $VK_UP
    Shot $hwnd "7_history_usage"
    Post-Key $hwnd $VK_DOWN 2

    # 7) 로그 필터: 글자
    Post-Click $hwnd $FilterX $FilterY
    Post-Text $hwnd "temp preview"
    Shot $hwnd "8_filter_word"
    Post-Key $hwnd $VK_BACK 12

    # 8) 로그 필터: 제외
    Post-Text $hwnd "-[Core], -[memory]"
    Shot $hwnd "9_filter_exclude"
    Post-Key $hwnd $VK_BACK 18

    # 9) 레벨: Info 끄기 → 경고 · 오류만
    Post-Click $hwnd $InfoX $InfoY
    Shot $hwnd "10_level"
}

[Win32PreviewCheck]::PostMessage($hwnd, 0x0010, [IntPtr]::Zero, [IntPtr]::Zero) | Out-Null   # WM_CLOSE
if (-not $proc.WaitForExit(40000)) { Write-Output "Timeout waiting exit, killing"; $proc.Kill(); exit 5 }
Write-Output "ExitCode $($proc.ExitCode)"
exit 0
