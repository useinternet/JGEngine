# DevConsole 창 입력 확인(DevConsole_TODO 3-4).
# 키 입력은 런처 창에만 PostMessage(WM_CHAR / WM_KEYDOWN / WM_KEYUP)로 보낸다. 전경 창을 바꾸지 않으므로 다른 창에 키가 들어가지 않는다.
# 엔진 창 프로시저는 모든 메시지를 ImGui_ImplWin32_WndProcHandler 로 넘긴다(DX12GUIBackend.cpp OnWndProc).

Add-Type @"
using System;
using System.Runtime.InteropServices;
public class Win32UiCheck {
    [StructLayout(LayoutKind.Sequential)]
    public struct RECT { public int Left; public int Top; public int Right; public int Bottom; }
    [DllImport("user32.dll")] public static extern bool GetWindowRect(IntPtr hWnd, out RECT lpRect);
    [DllImport("user32.dll")] public static extern bool PrintWindow(IntPtr hwnd, IntPtr hdcBlt, uint nFlags);
    [DllImport("user32.dll")] public static extern bool PostMessage(IntPtr hWnd, uint Msg, IntPtr wParam, IntPtr lParam);
    [StructLayout(LayoutKind.Sequential)]
    public struct POINT { public int X; public int Y; }
    [DllImport("user32.dll")] public static extern bool ClientToScreen(IntPtr hWnd, ref POINT lpPoint);
}
"@

function Post-Click([IntPtr]$hwnd, [int]$windowX, [int]$windowY) {
    # 창 좌표(캡처 기준) → 클라이언트 좌표
    $rect = New-Object Win32UiCheck+RECT
    [Win32UiCheck]::GetWindowRect($hwnd, [ref]$rect) | Out-Null
    $origin = New-Object Win32UiCheck+POINT
    [Win32UiCheck]::ClientToScreen($hwnd, [ref]$origin) | Out-Null
    $x = $windowX - ($origin.X - $rect.Left)
    $y = $windowY - ($origin.Y - $rect.Top)
    $lParam = [IntPtr](($y -shl 16) -bor ($x -band 0xFFFF))
    # 세 메시지를 잇달아 넣는다. 실제 커서가 창 밖이면 ImGui 백엔드의 TrackMouseEvent 가 곧 WM_MOUSELEAVE 를 보내므로,
    # 그보다 먼저 누름·뗌이 큐에 들어가 있어야 클릭 위치가 유지된다.
    [Win32UiCheck]::PostMessage($hwnd, 0x0200, [IntPtr]::Zero, $lParam) | Out-Null   # WM_MOUSEMOVE
    [Win32UiCheck]::PostMessage($hwnd, 0x0201, [IntPtr]1, $lParam) | Out-Null         # WM_LBUTTONDOWN (MK_LBUTTON)
    [Win32UiCheck]::PostMessage($hwnd, 0x0202, [IntPtr]::Zero, $lParam) | Out-Null    # WM_LBUTTONUP
    Start-Sleep -Milliseconds 500
    Write-Output "Clicked client ($x, $y)"
}

$WM_KEYDOWN = 0x0100
$WM_KEYUP   = 0x0101
$WM_CHAR    = 0x0102
$VK_RETURN  = 0x0D
$VK_UP      = 0x26
$VK_DOWN    = 0x28

function Save-Capture([IntPtr]$hwnd, [string]$path) {
    $rect = New-Object Win32UiCheck+RECT
    [Win32UiCheck]::GetWindowRect($hwnd, [ref]$rect) | Out-Null
    $width  = $rect.Right - $rect.Left
    $height = $rect.Bottom - $rect.Top
    $bmp = New-Object System.Drawing.Bitmap($width, $height)
    $gfx = [System.Drawing.Graphics]::FromImage($bmp)
    $hdc = $gfx.GetHdc()
    $ok = [Win32UiCheck]::PrintWindow($hwnd, $hdc, 2)
    $gfx.ReleaseHdc($hdc)
    $gfx.Dispose()
    if (-not $ok) { Write-Output "PrintWindow failed" }
    $bmp.Save($path, [System.Drawing.Imaging.ImageFormat]::Png)
    $bmp.Dispose()
    Write-Output "Saved $path"
}

function Post-Key([IntPtr]$hwnd, [int]$vk) {
    [Win32UiCheck]::PostMessage($hwnd, $WM_KEYDOWN, [IntPtr]$vk, [IntPtr]1) | Out-Null
    Start-Sleep -Milliseconds 80
    [Win32UiCheck]::PostMessage($hwnd, $WM_KEYUP, [IntPtr]$vk, [IntPtr]0xC0000001L) | Out-Null
    Start-Sleep -Milliseconds 120
}

function Post-Text([IntPtr]$hwnd, [string]$text) {
    foreach ($c in $text.ToCharArray()) {
        [Win32UiCheck]::PostMessage($hwnd, $WM_CHAR, [IntPtr][int]$c, [IntPtr]1) | Out-Null
        Start-Sleep -Milliseconds 30
    }
}

function Post-Line([IntPtr]$hwnd, [string]$text) {
    Post-Text $hwnd $text
    Post-Key $hwnd $VK_RETURN
    Start-Sleep -Milliseconds 700
}
