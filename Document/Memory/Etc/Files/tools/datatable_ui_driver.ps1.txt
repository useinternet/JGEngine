param(
    [Parameter(Mandatory = $true)][string]$Command,
    [string]$Arg1 = "",
    [string]$Arg2 = "",
    [string]$Arg3 = ""
)
# 데이터 테이블 편집 창 실제 입력 검증(V2) 도우미. 실행 중인 런처 창(PID 파일)에 메시지를 넣고 캡처한다.
# 사용자의 실제 마우스 · 키보드 · 전경 창은 건드리지 않는다: PostMessage 만 쓰고, Ctrl · Shift 는
# 대상 UI 스레드에 입력 큐를 붙여(AttachThreadInput) 키 상태만 잠시 바꾼다(ImGui 는 수정 키를 GetKeyState 로 읽는다).
#   start <BinDir>           런처를 띄우고 창을 기다린다(PID 를 state 파일에)
#   capture <png>            클라이언트 영역 캡처(PrintWindow)
#   click <x> <y> [right|double|shift]   클라이언트 좌표 클릭
#   drag <x1,y1> <x2,y2>     왼쪽 끌기
#   key <name> [ctrl|shift|ctrlshift]   Enter Tab Escape Back Delete Up Down Left Right Home End PageUp PageDown F2 Space 또는 A-Z 0-9 = -
#   text <string>            WM_CHAR(UTF-16) — IME 확정 글자와 같은 길
#   close                    WM_CLOSE 후 종료 코드
$ErrorActionPreference = "Stop"
Add-Type -AssemblyName System.Drawing
Add-Type @"
using System;
using System.Runtime.InteropServices;
public class DTUI {
    [StructLayout(LayoutKind.Sequential)] public struct RECT { public int Left; public int Top; public int Right; public int Bottom; }
    [DllImport("user32.dll")] public static extern bool GetClientRect(IntPtr hWnd, out RECT lpRect);
    [DllImport("user32.dll")] public static extern bool PrintWindow(IntPtr hwnd, IntPtr hdcBlt, uint nFlags);
    [DllImport("user32.dll", CharSet = CharSet.Unicode)] public static extern bool PostMessage(IntPtr hWnd, uint Msg, IntPtr wParam, IntPtr lParam);
    [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);
    [DllImport("kernel32.dll")] public static extern uint GetCurrentThreadId();
    [DllImport("user32.dll")] public static extern bool AttachThreadInput(uint idAttach, uint idAttachTo, bool fAttach);
    [DllImport("user32.dll")] public static extern bool GetKeyboardState(byte[] lpKeyState);
    [DllImport("user32.dll")] public static extern bool SetKeyboardState(byte[] lpKeyState);
    [DllImport("user32.dll")] public static extern uint MapVirtualKey(uint uCode, uint uMapType);
}
"@

$stateFile = Join-Path $PSScriptRoot "dt_ui_state.txt"

function Get-Hwnd {
    $procId = [int](Get-Content $stateFile)
    $proc = Get-Process -Id $procId -ErrorAction Stop
    if ($proc.MainWindowHandle -eq [IntPtr]::Zero) { throw "no main window" }
    return $proc.MainWindowHandle
}

function LParam([int]$x, [int]$y) {
    return [IntPtr](($y -shl 16) -bor ($x -band 0xFFFF))
}

function With-Modifiers([IntPtr]$hwnd, [string]$mods, [scriptblock]$action) {
    if ($mods -eq "") { & $action; return }
    $procIdOut = 0
    $target = [DTUI]::GetWindowThreadProcessId($hwnd, [ref]$procIdOut)
    $me = [DTUI]::GetCurrentThreadId()
    [DTUI]::AttachThreadInput($me, $target, $true) | Out-Null
    try {
        $state = New-Object byte[] 256
        [DTUI]::GetKeyboardState($state) | Out-Null
        if ($mods -like "*ctrl*") { $state[0x11] = 0x80; $state[0xA2] = 0x80 }
        if ($mods -like "*shift*") { $state[0x10] = 0x80; $state[0xA0] = 0x80 }
        [DTUI]::SetKeyboardState($state) | Out-Null
        & $action
        Start-Sleep -Milliseconds 400
        $state[0x11] = 0; $state[0xA2] = 0; $state[0x10] = 0; $state[0xA0] = 0
        [DTUI]::SetKeyboardState($state) | Out-Null
        # ImGui 는 키 이벤트가 올 때만 수정 키를 다시 읽는다 — 뗌 이벤트를 보내 Ctrl · Shift 가 눌린 채 남지 않게
        if ($mods -like "*ctrl*") { [DTUI]::PostMessage($hwnd, 0x0101, [IntPtr]0x11, [IntPtr]([int64]0xC01D0001)) | Out-Null }
        if ($mods -like "*shift*") { [DTUI]::PostMessage($hwnd, 0x0101, [IntPtr]0x10, [IntPtr]([int64]0xC02A0001)) | Out-Null }
        Start-Sleep -Milliseconds 300
    }
    finally {
        [DTUI]::AttachThreadInput($me, $target, $false) | Out-Null
    }
}

$vkMap = @{ "Enter" = 0x0D; "Tab" = 0x09; "Escape" = 0x1B; "Back" = 0x08; "Delete" = 0x2E; "Up" = 0x26; "Down" = 0x28; "Left" = 0x25; "Right" = 0x27;
            "Home" = 0x24; "End" = 0x23; "PageUp" = 0x21; "PageDown" = 0x22; "F2" = 0x71; "Space" = 0x20; "=" = 0xBB; "-" = 0xBD }

switch ($Command) {
    "start" {
        # 감시 프로세스가 런처를 띄우고 끝날 때까지 기다려 종료 코드를 남긴다(이 셸이 끝나도). 환경 변수는 이어받는다(JG_UITEST)
        $exitFile = Join-Path $PSScriptRoot "dt_ui_exit.txt"
        Remove-Item $exitFile -ErrorAction SilentlyContinue
        $exe = Join-Path $Arg1 "JGLauncher.exe"
        $watch = "`$p = Start-Process -FilePath '$exe' -WorkingDirectory '$Arg1' -PassThru; Set-Content -Path '$stateFile' -Value `$p.Id; `$p.WaitForExit(); Set-Content -Path '$exitFile' -Value `$p.ExitCode"
        Start-Process -FilePath "powershell.exe" -ArgumentList "-NoProfile", "-WindowStyle", "Hidden", "-Command", $watch | Out-Null
        $procId = 0
        for ($i = 0; $i -lt 120; $i++) {
            Start-Sleep -Milliseconds 500
            if ((Test-Path $stateFile) -and ((Get-Item $stateFile).LastWriteTime -gt (Get-Date).AddSeconds(-70))) {
                $procId = [int](Get-Content $stateFile)
                $proc = Get-Process -Id $procId -ErrorAction SilentlyContinue
                if ($proc -ne $null -and $proc.MainWindowHandle -ne [IntPtr]::Zero) { break }
            }
        }
        $proc = Get-Process -Id $procId
        $rect = New-Object DTUI+RECT
        [DTUI]::GetClientRect($proc.MainWindowHandle, [ref]$rect) | Out-Null
        Write-Output "PID $procId hwnd $($proc.MainWindowHandle) client $($rect.Right)x$($rect.Bottom) title '$($proc.MainWindowTitle)'"
    }    "capture" {
        $hwnd = Get-Hwnd
        $rect = New-Object DTUI+RECT
        [DTUI]::GetClientRect($hwnd, [ref]$rect) | Out-Null
        $bmp = New-Object System.Drawing.Bitmap($rect.Right, $rect.Bottom)
        $gfx = [System.Drawing.Graphics]::FromImage($bmp)
        $hdc = $gfx.GetHdc()
        [DTUI]::PrintWindow($hwnd, $hdc, 3) | Out-Null   # PW_CLIENTONLY | PW_RENDERFULLCONTENT
        $gfx.ReleaseHdc($hdc); $gfx.Dispose()
        $bmp.Save($Arg1, [System.Drawing.Imaging.ImageFormat]::Png)
        $bmp.Dispose()
        Write-Output "captured $Arg1 ($($rect.Right)x$($rect.Bottom))"
    }
    "click" {
        $hwnd = Get-Hwnd
        $x = [int]$Arg1; $y = [int]$Arg2
        $lp = LParam $x $y
        $mods = if ($Arg3 -eq "shift") { "shift" } else { "" }
        # 이동 · 누름 · 뗌을 쉬지 않고 넣는다. ImGui Win32 백엔드는 버튼 메시지의 좌표를 쓰지 않고 마지막 이동 위치를 쓰는데,
        # 사이에 쉬면 실제 커서가 창 밖이라 WM_MOUSELEAVE 가 끼어 위치가 실제 커서로 돌아간다. ImGui 는 큐를 프레임마다 나눠 처리한다.
        With-Modifiers $hwnd $mods {
            [DTUI]::PostMessage($hwnd, 0x0200, [IntPtr]::Zero, $lp) | Out-Null           # WM_MOUSEMOVE
            if ($Arg3 -eq "right") {
                [DTUI]::PostMessage($hwnd, 0x0204, [IntPtr]2, $lp) | Out-Null              # WM_RBUTTONDOWN
                [DTUI]::PostMessage($hwnd, 0x0205, [IntPtr]::Zero, $lp) | Out-Null         # WM_RBUTTONUP
            }
            else {
                [DTUI]::PostMessage($hwnd, 0x0201, [IntPtr]1, $lp) | Out-Null              # WM_LBUTTONDOWN
                [DTUI]::PostMessage($hwnd, 0x0202, [IntPtr]::Zero, $lp) | Out-Null         # WM_LBUTTONUP
                if ($Arg3 -eq "double") {
                    [DTUI]::PostMessage($hwnd, 0x0200, [IntPtr]::Zero, $lp) | Out-Null
                    [DTUI]::PostMessage($hwnd, 0x0203, [IntPtr]1, $lp) | Out-Null          # WM_LBUTTONDBLCLK
                    [DTUI]::PostMessage($hwnd, 0x0202, [IntPtr]::Zero, $lp) | Out-Null
                }
            }
        }
        Start-Sleep -Milliseconds 700
        Write-Output "click $x,$y $Arg3"
    }    "drag" {
        $hwnd = Get-Hwnd
        $p1 = $Arg1.Split(","); $p2 = $Arg2.Split(",")
        $lp1 = LParam ([int]$p1[0]) ([int]$p1[1]); $lp2 = LParam ([int]$p2[0]) ([int]$p2[1])
        [DTUI]::PostMessage($hwnd, 0x0200, [IntPtr]::Zero, $lp1) | Out-Null
        Start-Sleep -Milliseconds 100
        [DTUI]::PostMessage($hwnd, 0x0201, [IntPtr]1, $lp1) | Out-Null
        Start-Sleep -Milliseconds 150
        for ($s = 1; $s -le 6; $s++) {
            $x = [int]$p1[0] + ([int]$p2[0] - [int]$p1[0]) * $s / 6
            $y = [int]$p1[1] + ([int]$p2[1] - [int]$p1[1]) * $s / 6
            [DTUI]::PostMessage($hwnd, 0x0200, [IntPtr]1, (LParam ([int]$x) ([int]$y))) | Out-Null   # MK_LBUTTON
            Start-Sleep -Milliseconds 80
        }
        [DTUI]::PostMessage($hwnd, 0x0202, [IntPtr]::Zero, $lp2) | Out-Null
        Start-Sleep -Milliseconds 600
        Write-Output "drag $Arg1 -> $Arg2"
    }
    "key" {
        $hwnd = Get-Hwnd
        $vk = if ($vkMap.ContainsKey($Arg1)) { $vkMap[$Arg1] } else { [int][char]$Arg1.ToUpper() }
        $scan = [DTUI]::MapVirtualKey([uint32]$vk, 0)
        $down = [int64](1 -bor ($scan -shl 16))
        $up = [int64](1 -bor ($scan -shl 16) -bor 0xC0000000)
        With-Modifiers $hwnd $Arg2 {
            [DTUI]::PostMessage($hwnd, 0x0100, [IntPtr]$vk, [IntPtr]$down) | Out-Null   # WM_KEYDOWN
            Start-Sleep -Milliseconds 120
            [DTUI]::PostMessage($hwnd, 0x0101, [IntPtr]$vk, [IntPtr]$up) | Out-Null     # WM_KEYUP
        }
        Start-Sleep -Milliseconds 500
        Write-Output "key $Arg1 $Arg2"
    }
    "text" {
        $hwnd = Get-Hwnd
        # "u:D55C,AE00" = 코드포인트(16진) 목록 — 셸 인자로 한글이 깨지지 않게
        $chars = if ($Arg1.StartsWith("u:")) { $Arg1.Substring(2).Split(",") | ForEach-Object { [char][Convert]::ToInt32($_, 16) } } else { $Arg1.ToCharArray() }
        foreach ($ch in $chars) {
            [DTUI]::PostMessage($hwnd, 0x0102, [IntPtr][int]$ch, [IntPtr]1) | Out-Null   # WM_CHAR
            Start-Sleep -Milliseconds 40
        }
        Start-Sleep -Milliseconds 500
        Write-Output "text $($chars.Count) chars"
    }
    "release" {
        $hwnd = Get-Hwnd
        With-Modifiers $hwnd "ctrlshift" { }
        Write-Output "released modifiers"
    }
    "close" {
        $procId = [int](Get-Content $stateFile)
        $proc = Get-Process -Id $procId
        [DTUI]::PostMessage($proc.MainWindowHandle, 0x0010, [IntPtr]::Zero, [IntPtr]::Zero) | Out-Null   # WM_CLOSE
        $exitFile = Join-Path $PSScriptRoot "dt_ui_exit.txt"; for ($i = 0; $i -lt 120 -and -not (Test-Path $exitFile); $i++) { Start-Sleep -Milliseconds 500 }; if (Test-Path $exitFile) { Write-Output "exit code $(Get-Content $exitFile)" } else { Write-Output "did not exit in 60 s" }
    }
}
